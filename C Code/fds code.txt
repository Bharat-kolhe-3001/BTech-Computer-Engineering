#include <stdio.h>
#include <string.h>

#define MAX 50

int top = -1;
char a[MAX];

void push(char p) {
    if (top == MAX - 1)
        printf("overflow");
    else {
        top++;
        a[top] = p;
    }
}

void pop(char *text1) {
    if (top == -1) {
        printf("underflow");
        *text1 = '0';
    } else {
        *text1 = a[top];
        top--;
    }
}

int is_pair(char c1, char c2) {
    if (c1 == '(' && c2 == ')')
    return 1;
    else if(c1 == '[' && c2 == ']')
    return 1;
    else if(c1 == '{' && c2 == '}')
        return 1;
    else
        return 0;
}

int is_balanced(char text[]) {
    int i;
    for (i = 0; i < strlen(text); i++)
    {
        if (text[i] == '(' || text[i] == '[' || text[i] == '{')
        {
            push(text[i]);
        } 
        else if (text[i] == ')' || text[i] == ']' || text[i] == '}') 
        {
            if (top == -1)
                return 0;
            char text1;
            pop(&text1);
            if (!is_pair(text1, text[i]))
                return 0;
        }
    }
    if (top == -1)
        return 1;  // balanced
    else
        return 0;  // not balanced
}

int main()
{

3.queue
#include <stdio.h>
#define MAX 3

int queue[MAX];
int rear = -1;
int front = -1;

void en_queue()
{
    if (rear == MAX - 1)
    {
        printf("Queue is full\n");
    }
    else
    {
        int ele;
        printf("Enter element to enqueue: ");
        scanf("%d", &ele);
        rear++;
        queue[rear] = ele;
    }
}

void de_queue()
{
    if (front == rear)
    {
        printf("Queue is empty\n");
    }
    else
    {
        front++;
        int ele = queue[front];
        printf("Element deleted is %d\n", ele);
    }
}

void display()
{
    if (front == rear)
    {
        printf("Queue is empty\n");
    }
    else
    {
        printf("Queue elements are: ");
        for (int i = front + 1; i <= rear; i++)
        {
            printf("%d ", queue[i]);
        }
        printf("\n");
    }
}

int main()
{
    int choice, ch;

        default:
            printf("Invalid choice\n");
            break;
        }

}

4.circular queue
#include <stdio.h>
#define size 5

int queue[size];
int rear = -1;
int front = -1;
int ele;

void en_queue()
{
    if (rear == -1 && front == -1)
    {

        rear = 0;
        front = 0;
        queue[rear] = ele;
    }
    else if ((rear + 1) % size == front)
    {
        printf("Circular queue is full.\n");
    }
    else
    {
        rear = (rear + 1) % size;
        queue[rear] = ele;
    }
}

void de_queue()
{
    if (rear == -1 && front == -1)
    {
        printf("Circular Queue is empty\n");
    }
    else if (front == rear)
    {
        printf("Element deleted is %d\n", queue[front]);
        rear = -1;
        front = -1;
    }
    else
    {
        printf("Element deleted is %d\n", queue[front]);
        front = (front + 1) % size;
    }
}

void display()
{
    if (rear == -1 && front == -1)
    {
        printf("Circular Queue is empty\n");
    }
    else
    {
        int i = front;
        do
        {
            printf("%d\t", queue[i]);
            i = (i + 1) % size;
        } while (i != (rear + 1) % size);
        printf("\n");
    }
}

int main()
{
    int choice, ch;
    do
    {
        printf("1. Enqueue\n2. Dequeue\n3. Display\n");
        printf("Enter your choice: ");
        scanf("%d", &ch);

}

5.singly Linked list
#include <stdio.h>
#include <stdlib.h>

struct node
{
    int data;
    struct node *next;
};

struct node *head = NULL;

void insertAtFirst()
{
    struct node *newNode = malloc(sizeof(struct node));

    printf("Enter the element you want to insert at the first position: ");
    scanf("%d", &newNode->data);

    newNode->next = head;
    head = newNode;
}

void insertAtLast()
{
    struct node *newNode = malloc(sizeof(struct node));
    struct node *temp = head;

    printf("Enter the element you want to insert at the last position: ");
    scanf("%d", &newNode->data);

    newNode->next = NULL;

    if (head == NULL)
    {
        head = newNode;
        return;
    }

    while (temp->next != NULL)
    {
        temp = temp->next;
    }

    temp->next = newNode;
}

void insertAfterPosition()
{
    struct node *newNode = malloc(sizeof(struct node));
    struct node *temp = head;
    int pos, i = 1;

    printf("Enter the position after which you want to insert the new element: ");
    scanf("%d", &pos);

    while (i < pos && temp != NULL)
    {
        temp = temp->next;
        i++;
    }

    if (temp == NULL)
    {
        printf("Invalid position\n");
        return;
    }

    printf("Enter data: ");
    scanf("%d", &newNode->data);

    newNode->next = temp->next;
    temp->next = newNode;
}

void deleteFirst()
{
    if (head == NULL)
    {
        printf("Singly linked list is empty\n");
        return;
    }

    struct node *temp = head;
    head = head->next;
    free(temp);
}

void deleteLast()
{
    if (head == NULL)
    {
        printf("Singly linked list is empty\n");
        return;
    }

    struct node *temp = head;
    struct node *prevNode = NULL;

    while (temp->next != NULL)
    {
        prevNode = temp;
        temp = temp->next;
    }

    if (prevNode != NULL)
    {
        prevNode->next = NULL;
    }
    else
    {
        head = NULL;
    }

    free(temp);
}

void deleteAtPosition()
{
    if (head == NULL)
    {
        printf("Singly linked list is empty\n");
        return;
    }

    struct node *temp = head;
    struct node *prevNode = NULL;
    int pos, i = 1;

    printf("Enter the position from which you want to delete the node: ");
    scanf("%d", &pos);

    if (pos == 1)
    {
        head = temp->next;
        free(temp);
        return;
    }

    while (i < pos && temp != NULL)
    {
        prevNode = temp;
        temp = temp->next;
        i++;
    }

    if (temp == NULL)
    {
        printf("Invalid position\n");
        return;
    }

    prevNode->next = temp->next;

        free(head);
        head = NULL;
        return;
    }

    struct node *temp = head;
    while (temp->next != NULL) {
        temp = temp->next;
    }

    temp->prev->next = NULL;
    free(temp);

    printf("Enter the position from where you want to delete: ");
    scanf("%d", &pos);

    if (head == NULL) {
        printf("The list is empty\n");
        return;

    if (pos == 1) {
        deleteFirst();
        return;
    }

    struct node *temp = head;
    while (temp != NULL && i < pos) {
        temp = temp->next;
        i++;
    }

    if (temp == NULL) {
        printf("Invalid position\n");
        return;
    }

    
    if (temp->prev != NULL) {
        temp->prev->next = temp->next;
    }
    if (temp->next != NULL) {
        temp->next->prev = temp->prev;
    }
    free(temp);
}

        return;

    printf("data points: ");
    while (temp != NULL) {
        printf("%d ", temp->data);
        temp = temp->next;
    }
    printf("\n");
}

7.stack as a Linked list
#include <stdio.h>
#include <stdlib.h>

struct Node {
    int data;
    struct Node *next;
};

struct Node *top = NULL;

void push(int data) {
    struct Node *newnode = malloc(sizeof(struct Node));
    if (newnode == NULL) {
        printf("Memory allocation failed\n");
        return;
    }
    newnode->data = data;
    newnode->next = top;
    top = newnode;
}

void pop() {
    if (top == NULL) {
        printf("Underflow\n");
        return;
    }
    struct Node *temp = top;
    top = top->next;
    free(temp);
}

void display() {
    struct Node *temp = top;
    if (temp == NULL) {
        printf("Stack is empty\n");
        return;
    }
    printf("Stack elements: ");
    while (temp != NULL) {
        printf("%d ", temp->data);
        temp = temp->next;
    }
    printf("\n");
}

8.Queue as a Linked list
#include <stdio.h>
#include <stdlib.h>

struct Node 
{
    int data;
    struct Node *next;
};

struct Node *front = NULL, *rear = NULL, *newnode, *temp;

void enqueue(int data)
{
    newnode = malloc(sizeof(struct Node));
    newnode->data = data;
    newnode->next = NULL;
    
    if (front == NULL && rear == NULL)
    {
        front = newnode;
        rear = newnode;
    }
    else
    {
        rear->next = newnode;
        rear = newnode;
    }
}

void dequeue()
{
    if (front == NULL)
    {
        printf("Underflow\n");
    }
    else
    {
        temp = front;
        front = front->next;
        free(temp);
    }
}

void display()
{ 
    struct Node *temp = front;
    while (temp != NULL)
    { 
        printf("%d\t", temp->data);
        temp = temp->next;
    }
    printf("\n");
}

int main()
{
    int num_data_points, data, ch, choice;
